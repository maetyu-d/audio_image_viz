cmake_minimum_required(VERSION 3.20)
project(AudioImage VERSION 1.21.1 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
set(SUPPORT_FILEFORMAT_FLAC ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib" raylib EXCLUDE_FROM_ALL)
# Keep raylib's configured features but use a known analysis sample clock.
get_target_property(audioimage_raylib_definitions raylib COMPILE_DEFINITIONS)
list(FILTER audioimage_raylib_definitions EXCLUDE REGEX "^AUDIO_DEVICE_SAMPLE_RATE=")
list(FILTER audioimage_raylib_definitions EXCLUDE REGEX "^MAX_MESH_VERTEX_BUFFERS=")
set_property(TARGET raylib PROPERTY COMPILE_DEFINITIONS "${audioimage_raylib_definitions}")
target_compile_definitions(raylib PRIVATE AUDIO_DEVICE_SAMPLE_RATE=48000 MAX_MESH_VERTEX_BUFFERS=9)
add_library(audioimage_analysis src/analysis.cpp)
target_include_directories(audioimage_analysis PUBLIC src)
target_compile_options(audioimage_analysis PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  add_library(audioimage_streaming src/youtube.cpp src/youtube_json.mm src/spotify.mm)
  set_source_files_properties(src/spotify.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_include_directories(audioimage_streaming PUBLIC src)
  target_compile_options(audioimage_streaming PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(audioimage_streaming PRIVATE "-framework Foundation")
  add_executable(AudioImage MACOSX_BUNDLE src/main.cpp src/audio.cpp src/video.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/music.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/world.cpp")
  target_sources(AudioImage PRIVATE src/native.mm)
  set_target_properties(AudioImage PROPERTIES
    OUTPUT_NAME "AudioImage"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    MACOSX_BUNDLE_BUNDLE_NAME "AudioImage"
    MACOSX_BUNDLE_GUI_IDENTIFIER "studio.audioimage"
    MACOSX_BUNDLE_BUNDLE_VERSION "1.21.1"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "1.21.1")
  target_link_libraries(AudioImage PRIVATE audioimage_streaming "-framework AudioToolbox" "-framework AppKit" "-framework UniformTypeIdentifiers")
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/yt-dlp" OR NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/qjs")
    message(FATAL_ERROR "Run python3 scripts/fetch_youtube_tools.py before building the macOS app.")
  endif()
  add_custom_command(TARGET AudioImage POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:AudioImage>/../Helpers" "$<TARGET_FILE_DIR:AudioImage>/../Resources/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/yt-dlp" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/qjs" "$<TARGET_FILE_DIR:AudioImage>/../Helpers"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/yt-dlp-LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/yt-dlp-THIRD_PARTY_LICENSES.txt" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/QuickJS-LICENSE" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/GPL-3.0.txt" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/youtube/versions.json" "$<TARGET_FILE_DIR:AudioImage>/../Resources/licenses")
  add_custom_command(TARGET AudioImage POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/vendor/raylib/LICENSE" "$<TARGET_FILE_DIR:AudioImage>/../Resources/licenses/raylib-LICENSE")
else()
  add_executable(AudioImage src/main.cpp src/audio.cpp src/video.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/music.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/world.cpp")
endif()
target_include_directories(AudioImage PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(AudioImage PRIVATE audioimage_analysis raylib)
target_compile_options(AudioImage PRIVATE -Wall -Wextra -Wpedantic)
if(APPLE)
  add_custom_command(TARGET AudioImage POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts" "$<TARGET_FILE_DIR:AudioImage>/../Resources/assets/fonts")
else()
  add_custom_command(TARGET AudioImage POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts" "$<TARGET_FILE_DIR:AudioImage>/assets/fonts")
endif()
include(CTest)

  if(APPLE)

    target_link_libraries(spotify_tests PRIVATE audioimage_streaming)

  add_executable(analysis_tests tests/analysis.cpp)
  target_link_libraries(analysis_tests PRIVATE audioimage_analysis)

  add_executable(synth_analysis_tests tests/synth_analysis.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/music.cpp")
  target_include_directories(synth_analysis_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(synth_analysis_tests PRIVATE audioimage_analysis)

  target_include_directories(audio_playback_tests PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(audio_playback_tests PRIVATE audioimage_analysis raylib)
  if(APPLE)
    target_link_libraries(audio_playback_tests PRIVATE "-framework AudioToolbox")

  add_executable(audio_sample_rate_tests tests/audio_sample_rates.cpp src/audio.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/music.cpp")
  target_include_directories(audio_sample_rate_tests PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(audio_sample_rate_tests PRIVATE audioimage_analysis raylib)
  if(APPLE)
    target_link_libraries(audio_sample_rate_tests PRIVATE "-framework AudioToolbox")

  add_executable(scene_render_tests tests/video_scenes.cpp src/video.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/world.cpp")

  target_include_directories(memory_render_tests PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(memory_render_tests PRIVATE audioimage_analysis raylib)
  # Explicit native check: requires a desktop OpenGL context, so it is run
  # separately from the device-free CTest suite.
  add_executable(reaction_render_tests tests/video_reaction.cpp src/video.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/world.cpp")
  target_include_directories(reaction_render_tests PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(reaction_render_tests PRIVATE audioimage_analysis raylib)
  add_executable(video_review tests/video_review.cpp src/video.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/world.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/music.cpp")
  target_include_directories(video_review PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_link_libraries(video_review PRIVATE audioimage_analysis raylib)
  if(APPLE)
    add_executable(video_benchmark tests/video_benchmark.cpp src/video.cpp "${CMAKE_CURRENT_SOURCE_DIR}/src/world.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/music.cpp")
    target_include_directories(video_benchmark PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_link_libraries(video_benchmark PRIVATE audioimage_analysis raylib)
  endif()
endif()
